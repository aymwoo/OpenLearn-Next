```typescript
export async function executeLessonDraftPersist(ctx, command) {
  // 1) 授权：authorizeLessonDraftCommand(ctx, command)
  //    → assertActiveTeacher + 校验 teacher.schoolId === lesson.schoolId（越权写他校 draft 拦截，T-63-01）
  const teacher = await authorizeLessonDraftCommand(ctx, command);

  // 2) 调 DAL（Plan 02）；teacherId 经闭包/ctx，不读 payload；sourceCommandId = command.id
  const { draftVersionId, version, stepCount } = await persistDraftLessonVersion({
    lessonId: command.payload.lessonId,
    steps: command.payload.steps,
    sourceCommandId: command.id,          // provenance + 表层幂等键（dedupeKey→command 复用同 id）
    createdById: teacher.id,              // 起草教师 id 注入表 createdById（闭包注入，非 payload）
  });

  // 3) 构造 summary-only 事件（Plan 03 schema）
  const event = {
    eventType: "lesson.draft.persisted",
    category: "domain", aggregateType: "lesson", aggregateId: command.payload.lessonId,
    payload: { draftVersionId, version, stepCount, source: "ai" },
    audit: { delegatedActor: null, approval: null },
  };

  // 4) 返回 bus 约定形状；invalidation.tags 用 cacheTags（不在 execute 内直接 updateTag，由 bus 统一失效）
  return {
    result: { draftVersionId, version, stepCount },
    events: [event],
    invalidation: { tags: [cacheTags.draftLesson(lessonId), cacheTags.lesson(lessonId)] },
  };
}
```

约束：
- **不**在 execute 内调用 `updateTag()`（写后失效由 bus 依据返回的 invalidation.tags 统一处理，符合 STACK caching 约定）。
- **不**在 persist handler 内调用 `lesson.draft.run` / AI 生成（D-05 解耦）。
- teacherId/source 绝不来自 payload；source 已在 DAL 硬编码 'ai'。
- import：`persistDraftLessonVersion`（@/lib/dal/lesson-authoring）、`cacheTags`（@/lib/cache-policy）。

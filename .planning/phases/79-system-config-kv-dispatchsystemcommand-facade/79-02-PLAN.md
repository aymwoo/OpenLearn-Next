**matchConfigKey 函数**（D-11，镜像 matchDomain line 84-95）:
```typescript
function matchConfigKey(pattern: string, key: string): boolean {
  if (pattern.endsWith(":*")) {
    const prefix = pattern.slice(0, -2);
    return key.startsWith(prefix + ":") && !key.slice(prefix.length + 1).includes(":");
  }
  return pattern === key;
}
```

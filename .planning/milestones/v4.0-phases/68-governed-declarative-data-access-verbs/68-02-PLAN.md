async function assertTeacherManagerScope(actorId, schoolId) {
  if (!actorId?.trim()) throw new Error("PLUGIN_ACTOR_REQUIRED");
  const scope = await assertActiveTeacher();       // { userId, schoolIds }
  if (scope.userId !== actorId || !scope.schoolIds.includes(schoolId)) throw new Error("TEACHER_AUTH_REQUIRED");
  return scope;
}
